Code Input
cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
results = {}  # store model results

def evaluate_model(name, model, X_tr, y_tr, X_te, y_te):
    """Train, predict, and store metrics."""
    model.fit(X_tr, y_tr)
    y_pred = model.predict(X_te)
    y_proba = model.predict_proba(X_te)[:, 1] if hasattr(model, 'predict_proba') else None
    
    f1 = f1_score(y_te, y_pred)
    prec = precision_score(y_te, y_pred)
    rec = recall_score(y_te, y_pred)
    auc = roc_auc_score(y_te, y_proba) if y_proba is not None else None
    
    results[name] = {'F1': f1, 'Precision': prec, 'Recall': rec, 'ROC-AUC': auc}
    print(f"\n{'='*50}")
    print(f"{name}")
    print(f"{'='*50}")
    print(f"F1: {f1:.4f} | Precision: {prec:.4f} | Recall: {rec:.4f} | AUC: {auc:.4f}" if auc else f"F1: {f1:.4f}")
    print(classification_report(y_te, y_pred, target_names=['Not Request', 'Request']))
    
    # Confusion matrix
    fig, ax = plt.subplots(figsize=(5, 4))
    ConfusionMatrixDisplay.from_predictions(y_te, y_pred, display_labels=['Not Request', 'Request'], ax=ax, cmap='Blues')
    ax.set_title(f'{name} — Confusion Matrix')
    plt.tight_layout()
    plt.show()
    
    return model

def predict_message(message, model=xgb_model, model_name='XGBoost'):
    """Predict whether a message is a request for help."""
    # Extract TF-IDF features
    tfidf_feat = tfidf.transform([message])
    
    # Extract metadata features
    meta = pd.DataFrame([{
        'msg_len': len(message),
        'word_count': len(message.split()),
        'genre_direct': 1,  # assume direct message
        'genre_news': 0,
        'genre_social': 0,
        'has_question': int('?' in message),
        'has_exclaim': int('!' in message),
        'upper_ratio': sum(1 for c in message if c.isupper()) / max(len(message), 1)
    }])
    meta_scaled = scaler.transform(meta)
    
    # Combine features
    X_new = hstack([tfidf_feat, csr_matrix(meta_scaled)])
    
    # Predict
    pred = model.predict(X_new)[0]
    proba = model.predict_proba(X_new)[0]
    
    print(f"\n{'='*60}")
    print(f"LIVE PREDICTION ({model_name})")
    print(f"{'='*60}")
    print(f"Message: {message}")
    print(f"Prediction: {'REQUEST FOR HELP' if pred == 1 else 'NOT A REQUEST'}")
    print(f"Confidence: {proba[pred]:.2%}")
    print(f"P(Not Request): {proba[0]:.4f} | P(Request): {proba[1]:.4f}")
    return pred, proba
